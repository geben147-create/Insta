#!/usr/bin/env bash
# v04 "kitten knight x red dragon pt.2" re-assembly (26.00 s, 1080x1920, 30 fps, 780 frames)
# Inputs in the current folder:
#   s01.mp4  9:16 vertical gen (1080x1920 or 720x1280), >= 4.60 s   (kitten in fog + 90deg camera roll)
#   s02.mp4  16:9 LANDSCAPE gen (1920x1080 or 1280x720), >= 7.67 s  (dragon attack + hide behind rock)
#   s03.mp4  16:9 LANDSCAPE gen, >= 5.00 s                           (dragon scoops kitten, takes off)
#   s04.mp4  16:9 LANDSCAPE gen, >= 8.40 s                           (saddled dragon, wing unfurl, flyover, flies into fog)
#   bgm.mp3  epic cinematic track, bar downbeat hits every ~2.2 s (110 BPM), first big hit at 6.51 s
#   sfx/*.wav  (optional, see SFX list; missing files are skipped)
# ROLL_IN_EDIT=1 -> s01 was generated WITHOUT the roll: the edit adds the 0->85deg clockwise roll 3.93-4.60 s.
# ROLL_IN_EDIT=0 -> s01 already contains the roll (preferred, like the original).
# S2_IN / S3_IN / S4_IN = seconds to skip at the head of each gen (pick the best 7.67 / 5.10 / 8.63 s window).
set -euo pipefail
ROLL_IN_EDIT=${ROLL_IN_EDIT:-0}
S1_IN=${S1_IN:-0}; S2_IN=${S2_IN:-0}; S3_IN=${S3_IN:-0}; S4_IN=${S4_IN:-0}
OUT=${OUT:-v04_remake.mp4}

# ---------- video ----------
PREP_V='fps=30,scale=1080:1920:force_original_aspect_ratio=increase:flags=lanczos,crop=1080:1920,setsar=1'
PREP_L='fps=30,scale=1920:1080:force_original_aspect_ratio=increase:flags=lanczos,crop=1920:1080,transpose=1,setsar=1'
# transpose=1 = rotate 90deg CLOCKWISE -> ground ends up on the LEFT edge, sky on the right (exactly like the original)

if [ "$ROLL_IN_EDIT" = "1" ]; then
  # roll: theta(n)=1.48*(n/20)^2 rad (ease-in, 0 -> 85deg in 20 frames = 3.93-4.60 s). Frame is upscaled 2x, rotated, then zoompan crops
  # with zoom = cos(theta)+1.778*sin(theta) (the minimum that hides the black corners),
  # tmix=3 fakes rotational motion blur. Hard cut to the rotated s02 at 4.60 s completes the "world turned sideways" gag.
  V1="[0:v]trim=start=${S1_IN}:duration=4.6,setpts=PTS-STARTPTS,${PREP_V},trim=end_frame=138,split=2[a0][b0];
[a0]trim=end_frame=118,setpts=PTS-STARTPTS[a1];
[b0]trim=start_frame=118,setpts=PTS-STARTPTS,scale=2160:3840:flags=bicubic,rotate=a='1.48*pow(min((n+1)/20,1),2)':ow=2160:oh=3840:c=black,zoompan=z='1.02*(cos(1.48*pow(min((on+1)/20,1),2))+1.778*sin(1.48*pow(min((on+1)/20,1),2)))':x='iw/2-(iw/zoom/2)':y='ih/2-(ih/zoom/2)':d=1:s=1080x1920:fps=30,tmix=frames=3[b1];
[a1][b1]concat=n=2:v=1:a=0[v1];"
else
  V1="[0:v]trim=start=${S1_IN}:duration=4.8,setpts=PTS-STARTPTS,${PREP_V},trim=end_frame=138,setpts=PTS-STARTPTS[v1];"
fi

VF="${V1}
[1:v]trim=start=${S2_IN},setpts=PTS-STARTPTS,${PREP_L},tpad=stop_mode=clone:stop_duration=0.5,trim=end_frame=230,setpts=PTS-STARTPTS[v2];
[2:v]trim=start=${S3_IN},setpts=PTS-STARTPTS,${PREP_L},tpad=stop_mode=clone:stop_duration=0.5,trim=end_frame=153,setpts=PTS-STARTPTS[v3];
[3:v]trim=start=${S4_IN},setpts=PTS-STARTPTS,${PREP_L},tpad=stop_mode=clone:stop_duration=0.5,trim=end_frame=259,setpts=PTS-STARTPTS[v4];
[v1][v2][v3][v4]concat=n=4:v=1:a=0,eq=saturation=0.94:contrast=0.97:gamma=1.02,format=yuv420p[vout]"
# cut points: 4.600 (f138) | 12.267 (f368) | 17.367 (f521) | 26.000 (f780) -- all HARD cuts, no xfade.

# ---------- audio ----------
# file | start seconds | gain dB   (times measured from the original's onsets / loudness envelope; SFX identities are inferred)
SFX=(
"sfx/01_moor_wind_bed.wav|0.00|-14"
"sfx/02_soft_sub_boom.wav|0.25|-10"
"sfx/03_reverse_riser.wav|3.50|-9"
"sfx/04_roll_whoosh.wav|4.05|-6"
"sfx/05_dragon_low_growl.wav|4.60|-10"
"sfx/06_dragon_roar.wav|5.55|-3"
"sfx/07_camera_knock_thud.wav|6.25|-4"
"sfx/08_giant_footstep_1.wav|6.51|-5"
"sfx/08_giant_footstep_2.wav|6.78|-6"
"sfx/08_giant_footstep_3.wav|7.06|-6"
"sfx/09_armor_tumble_clatter.wav|7.40|-8"
"sfx/10_dirt_debris_patter.wav|8.15|-12"
"sfx/11_tail_swish.wav|9.20|-12"
"sfx/12_jaw_snap_growl.wav|9.65|-8"
"sfx/13_dragon_sniff_snort.wav|11.15|-6"
"sfx/14_head_swing_whoosh_clink.wav|13.05|-8"
"sfx/15_wing_flap_takeoff.wav|15.55|-5"
"sfx/16_wingbeat.wav|16.35|-9"
"sfx/17_landing_thud_flap.wav|17.37|-5"
"sfx/18_wing_unfurl_snap.wav|19.95|-4"
"sfx/19_takeoff_boom_whoosh.wav|21.45|-3"
"sfx/20_flyover_wind_gust.wav|22.00|-6"
"sfx/21_distant_wingbeats.wav|23.00|-14"
)
INPUTS=(-i s01.mp4 -i s02.mp4 -i s03.mp4 -i s04.mp4 -i bgm.mp3)
AF="[4:a]atrim=0:26,asetpts=PTS-STARTPTS,afade=t=in:d=0.05,aformat=sample_rates=48000:channel_layouts=stereo,volume='if(between(t,5.55,6.40)+between(t,21.45,22.30),0.71,1)':eval=frame[bgm];"
MIX="[bgm]"; N=1; IDX=5
for row in "${SFX[@]}"; do
  IFS='|' read -r f t g <<< "$row"
  [ -f "$f" ] || continue
  ms=$(python -c "print(int(round(float('$t')*1000)))")
  INPUTS+=(-i "$f")
  AF+="[${IDX}:a]aformat=sample_rates=48000:channel_layouts=stereo,adelay=${ms}|${ms},volume=${g}dB[s${IDX}];"
  MIX+="[s${IDX}]"; N=$((N+1)); IDX=$((IDX+1))
done
# bgm is ducked -3 dB under the roar (5.55-6.40 s) and the takeoff (21.45-22.30 s) by the volume expression above
AF+="${MIX}amix=inputs=${N}:normalize=0:duration=first,atrim=0:26,afade=t=out:st=25.97:d=0.03,alimiter=limit=0.89[aout]"

ffmpeg -y -hide_banner -v error "${INPUTS[@]}" \
  -filter_complex "${VF};${AF}" \
  -map "[vout]" -map "[aout]" -r 30 -frames:v 780 \
  -c:v libx264 -preset slow -crf 16 -profile:v high -pix_fmt yuv420p -movflags +faststart \
  -c:a aac -b:a 192k -ar 48000 -t 26.0 "$OUT" < /dev/null
echo "wrote $OUT"
