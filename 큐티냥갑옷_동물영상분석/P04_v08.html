7. FFmpeg 편집 스크립트

  • 1) 생성 클립 준비: s01.mp4(9:16 1080x1920), s02~s08.mp4(16:9 1920x1080), bgm.mp3, sfx/*.wav
  • 2) 각 클립을 지정 길이로 trim, s02~s08은 transpose=1(시계방향 90도)로 1080x1920 변환
  • 3) 전부 24fps로 통일해 concat(하드컷만 사용)
  • 4) 색보정(살짝 채도 낮추고 대비 부드럽게) + 필름 그레인
  • 5) 30fps로 변환(fps=30 → 24→30 중복 프레임, 원본과 동일한 케이던스)
  • 6) BGM + 효과음을 adelay로 타임라인에 배치 후 amix, 마지막 0.35s 페이드아웃
  • 7) libx264 CRF 17, AAC 192k로 최종 출력
전체 조립 스크립트 (s01.mp4… + bgm + sfx → 최종본)
#!/usr/bin/env bash
# v08 kitten-knight x dragon pt.3 reconstruction. Inputs: s01.mp4 (9:16), s02..s08.mp4 (16:9), bgm.mp3, sfx/*.wav
set -euo pipefail
OUT=final_v08.mp4
: > list.txt
# durations per shot (seconds)
D=(6.53 2.80 3.37 3.87 5.00 5.00 2.63 7.37)
V=""
for i in 1 2 3 4 5 6 7 8; do
  f=$(printf "s%02d.mp4" $i); d=${D[$((i-1))]}
  if [ $i -eq 1 ]; then
    ffmpeg -v error -y -i "$f" -t "$d" -vf "scale=1080:1920:force_original_aspect_ratio=increase,crop=1080:1920,fps=24,setsar=1" -an -c:v libx264 -crf 14 -pix_fmt yuv420p "p$i.mp4" < /dev/null
  else
    # 16:9 landscape rotated 90 degrees clockwise -> sky on the right, ground on the left
    ffmpeg -v error -y -i "$f" -t "$d" -vf "scale=1920:1080:force_original_aspect_ratio=increase,crop=1920:1080,transpose=1,fps=24,setsar=1" -an -c:v libx264 -crf 14 -pix_fmt yuv420p "p$i.mp4" < /dev/null
  fi
  echo "file 'p$i.mp4'" >> list.txt
done
ffmpeg -v error -y -f concat -safe 0 -i list.txt -c copy video_24.mp4 < /dev/null
# optional impact shake 21.57-21.90s if the crash gen lacks it
# grade + grain, then 24->30fps by frame duplication (matches source cadence)
ffmpeg -v error -y -i video_24.mp4 -vf "crop=iw-24:ih-42:12+6*sin(90*t)*between(t,21.57,21.90):21+8*cos(110*t)*between(t,21.57,21.90),scale=1080:1920,eq=saturation=0.95:contrast=0.97,noise=alls=4:allf=t,fps=30" -c:v libx264 -crf 16 -pix_fmt yuv420p video_30.mp4 < /dev/null
# audio: bgm + sfx at exact times (ms delays)
ffmpeg -v error -y -i video_30.mp4 -i bgm.mp3 \
 -i sfx/wind_bed.wav -i sfx/gust_wing.wav -i sfx/armor_rattle.wav -i sfx/growl_low.wav -i sfx/whoosh_wing.wav \
 -i sfx/wing_flap.wav -i sfx/wing_downstroke.wav -i sfx/wind_rush.wav -i sfx/dive_riser.wav -i sfx/roar_short.wav \
 -i sfx/wind_blast.wav -i sfx/grunt.wav -i sfx/impact.wav -i sfx/skid.wav -i sfx/debris.wav -i sfx/land_mew.wav \
 -i sfx/footsteps.wav -i sfx/eye_rumble.wav -i sfx/puff_chime.wav \
 -filter_complex "\
[1:a]volume=-18dB,atrim=0:36.57[bgm];\
[2:a]adelay=210|210,volume=-14dB,atrim=0:23[a2];\
[3:a]adelay=2510|2510,volume=-6dB[a3];\
[4:a]adelay=3250|3250,volume=-10dB[a4];\
[5:a]adelay=4480|4480,volume=-9dB[a5];\
[6:a]adelay=6530|6530,volume=-6dB[a6];\
[7:a]adelay=7060|7060,volume=-9dB[a7];\
[8:a]adelay=10900|10900,volume=-7dB[a8];\
[9:a]adelay=12700|12700,volume=-8dB[a9];\
[10:a]adelay=15310|15310,volume=-4dB[a10];\
[11:a]adelay=15700|15700,volume=-6dB[a11];\
[12:a]adelay=16570|16570,volume=-6dB[a12];\
[13:a]adelay=20750|20750,volume=-8dB[a13];\
[14:a]adelay=21490|21490,volume=-3dB[a14];\
[15:a]adelay=21700|21700,volume=-7dB[a15];\
[16:a]adelay=22050|22050,volume=-12dB[a16];\
[17:a]adelay=23100|23100,volume=-10dB[a17];\
[18:a]adelay=24060|24060,volume=-16dB[a18];\
[19:a]adelay=30900|30900,volume=-12dB[a19];\
[20:a]adelay=33900|33900,volume=-10dB[a20];\
[bgm][a2][a3][a4][a5][a6][a7][a8][a9][a10][a11][a12][a13][a14][a15][a16][a17][a18][a19][a20]amix=inputs=20:normalize=0,alimiter=limit=0.9,afade=t=out:st=36.22:d=0.35,atrim=0:36.57[aout]" \
 -map 0:v -map "[aout]" -c:v copy -c:a aac -b:a 192k -shortest "$OUT" < /dev/null
rm -f p?.mp4 list.txt video_24.mp4 video_30.mp4
echo "done -> $OUT"
